# Packages installed into the Airflow environment (on top of apache/airflow:2.10.5-python3.11).
#
# Only Airflow-side libraries go here. dbt + elementary-data are installed into a
# SEPARATE virtualenv (/opt/dbt_venv) in the Dockerfile, because dbt-core requires
# sqlparse>=0.5.5 which conflicts with the versions Airflow 2.10.5 ships. Keeping them
# isolated is the standard astronomer-cosmos pattern and avoids dependency deadlock.
#
# These are installed WITH Airflow's official 2.10.5 constraints file (see Dockerfile),
# which pins apache-airflow==2.10.5 and prevents pip from silently upgrading the core.
# Leave packages the constraints file manages (airbyte provider, snowflake-connector)
# unpinned so the tested-compatible version is selected.

# Airbyte provider (gives DAGs airflow.providers.airbyte.*)
apache-airflow-providers-airbyte

# Cosmos orchestrates dbt as a subprocess; it does NOT depend on dbt itself.
astronomer-cosmos==1.14.2

# Used by the sgdp platform services running inside Airflow.
snowflake-connector-python
gitpython
